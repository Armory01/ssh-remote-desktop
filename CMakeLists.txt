cmake_minimum_required(VERSION 3.15)
cmake_policy(SET CMP0091 NEW)
project(SSHRemoteDesktop VERSION 0.4.2 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "SSH Remote Desktop requires Windows and MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(ra_core STATIC src/common.cpp src/sessions.cpp src/bridge.cpp src/desktop.cpp src/system.cpp)
target_include_directories(ra_core PUBLIC src third_party)
target_compile_definitions(ra_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0601 WINVER=0x0601)
target_compile_options(ra_core PUBLIC /W4 /permissive- /utf-8 /EHsc)
target_link_libraries(ra_core PUBLIC advapi32 user32 gdi32 ole32 oleaut32 wtsapi32 taskschd windowscodecs shell32 uuid uiautomationcore netapi32)
add_executable(ra WIN32 src/main.cpp)
target_link_libraries(ra PRIVATE ra_core)
target_link_options(ra PRIVATE /SUBSYSTEM:WINDOWS,6.01)
include(CTest)
if(BUILD_TESTING)
  add_executable(ra_core_tests tests/core_tests.cpp)
  target_link_libraries(ra_core_tests PRIVATE ra_core)
  add_test(NAME core COMMAND ra_core_tests)
  add_executable(ra_gui_fixture WIN32 tests/gui_fixture.cpp)
  target_link_libraries(ra_gui_fixture PRIVATE ra_core)
  target_link_options(ra_gui_fixture PRIVATE /SUBSYSTEM:WINDOWS,6.01)
  add_executable(ra_test_askpass tests/askpass.cpp)
  target_compile_definitions(ra_test_askpass PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
endif()
